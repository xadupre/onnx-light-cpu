cmake_minimum_required(VERSION 3.15)
project(cpu_compute_consumer LANGUAGES CXX)

if(NOT TARGET cpu_compute::cpu_compute)
  find_package(cpu_compute CONFIG REQUIRED)
endif()
enable_testing()
add_executable(compute_consumer consumer.cc)
target_link_libraries(compute_consumer PRIVATE cpu_compute::cpu_compute)
add_test(NAME compute_consumer COMMAND compute_consumer)
